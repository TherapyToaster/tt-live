if(document.body&&document.body.classList.contains('sqs-edit-mode-active'))return;
var old=document.getElementById('tt-intro');if(old)old.remove();
document.documentElement.classList.remove('tt-intro-lock');
var box=root.querySelector('#tt-arc-enter'),v=root.querySelector('#tt-arc-intro'),
    un=root.querySelector('.ta-unmute'),go=root.querySelector('.ta-enter-go'),sk=root.querySelector('.ta-enter-skip'),
    floor=root.querySelector('#tt-arc-floor');
if(!box||!v)return;
var entered=0,RM=window.matchMedia&&window.matchMedia('(prefers-reduced-motion:reduce)');
function into(){
  if(entered)return;entered=1;
  try{v.pause()}catch(e){}
  if(floor&&!(RM&&RM.matches)){try{floor.scrollIntoView({behavior:'smooth',block:'start'})}catch(e){floor.scrollIntoView(true)}}
  else if(floor){try{floor.scrollIntoView({behavior:'auto',block:'start'})}catch(e){floor.scrollIntoView(true)}}
}
function hasAudio(){
  try{if(typeof v.mozHasAudio==='boolean')return v.mozHasAudio}catch(e){}
  try{if(v.audioTracks)return v.audioTracks.length>0}catch(e){}
  try{if(typeof v.webkitAudioDecodedByteCount==='number'&&v.currentTime>0.15)return v.webkitAudioDecodedByteCount>0}catch(e){}
  return null;
}
if(un){
  un.addEventListener('click',function(){
    var a=hasAudio();
    if(a===false){var prev=un.textContent;un.textContent='No audio';setTimeout(function(){un.textContent=prev},1400);return;}
    v.muted=!v.muted;un.setAttribute('aria-pressed',v.muted?'false':'true');
    un.textContent=v.muted?'Unmute':'Mute';
    un.setAttribute('aria-label',v.muted?'Unmute intro':'Mute intro');
    if(!v.muted){var p=v.play&&v.play();if(p&&p.catch)p.catch(function(){});}
  });
}
if(go)go.addEventListener('click',into);
if(sk)sk.addEventListener('click',into);
v.addEventListener('ended',into);
v.muted=true;
var p=v.play&&v.play();if(p&&p.catch)p.catch(function(){});
})();
</script>
